ch32h417qeu6.upload.maximum_size=933888
ch32h417qeu6.upload.maximum_data_size=917504

# Off by default, as in arduino-pico and the ESP cores. Enabling costs roughly
# 30 KB of unwind tables, which stay in flash and are read only when something
# throws. Note that --specs=nano.specs must never be added: it rewrites
# -lstdc++ to -lstdc++_nano, which has no .eh_frame, and every throw then
# reaches std::terminate after a perfectly clean link.
ch32h417qeu6.menu.exceptions.Disabled=Disabled
ch32h417qeu6.menu.exceptions.Disabled.build.exceptions=disabled
ch32h417qeu6.menu.exceptions.Enabled=Enabled
ch32h417qeu6.menu.exceptions.Enabled.build.exceptions=enabled

# What `Serial` refers to. Serial1 is always USART1 on PA9/PA10, into the
# WCH-Link's VCP, whichever is chosen here.
#
# USB CDC is the default because it is what someone plugging the board into a
# PC expects to find. The UART option is worth keeping for debugging: a fault
# during static initialisation happens before USB has enumerated, so a board
# that only speaks CDC cannot report one.
ch32h417qeu6.menu.serial.usb=USB CDC
ch32h417qeu6.menu.serial.usb.build.serial=usb
ch32h417qeu6.menu.serial.uart=USART1
ch32h417qeu6.menu.serial.uart.build.serial=uart
