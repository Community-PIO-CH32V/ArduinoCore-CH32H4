# Arduino IDE / arduino-cli build recipes for the CH32H41x core.
#
# PlatformIO does not read this file -- it uses tools/platformio-build.py, and
# the two are two statements of the same build. Where they can share, they do:
# both substitute the same token into the same linker-script template, and both
# derive the flash layout from the same constants.
#
# ### The SDK
#
# arduino-cli compiles cores/{build.core} and variants/{build.variant} and
# nothing else. The vendor SDK lives in system/ch32h417lib, outside both, so a
# prebuild hook compiles it into an archive in the build directory and the
# combine recipe links that. arduino-pico has the same problem and ships a
# prebuilt libpico.a; building it here instead keeps the repository free of
# committed binaries and means the SDK is always compiled with the same flags
# as everything else.
#
# ### Python
#
# Two hooks need it: the SDK archive and the linker-script substitution. The
# command is a property so it can be overridden in platform.local.txt without
# editing this file -- a published package would declare a python tool and
# point this at {runtime.tools.<tool>.path} instead.

name=CH32H41x Boards
version=1.0.0

# --------------------------------------------------------------------------
# Toolchain
# --------------------------------------------------------------------------
#
# The WCH RISC-V toolchain, GCC 12.2. NOT the riscv-none-embed-gcc 8.2 that
# WCH's own older Arduino package ships: this core needs -march with the Zb*
# bit-manipulation extensions and the xw compressed extension, and gnu++17.
#
# compiler.path is overridden in platform.local.txt for a local checkout. A
# published package would resolve it from a declared tool.
compiler.path={runtime.tools.riscv-wch-elf-gcc.path}/bin/
compiler.prefix=riscv-wch-elf-

ch32h4.python=python

compiler.c.cmd={compiler.prefix}gcc
compiler.cpp.cmd={compiler.prefix}g++
compiler.S.cmd={compiler.prefix}gcc
compiler.ar.cmd={compiler.prefix}gcc-ar
compiler.c.elf.cmd={compiler.prefix}g++
compiler.objcopy.cmd={compiler.prefix}objcopy
compiler.elf2hex.cmd={compiler.prefix}objcopy
compiler.size.cmd={compiler.prefix}size
compiler.objdump.cmd={compiler.prefix}objdump

# --------------------------------------------------------------------------
# Flags
# --------------------------------------------------------------------------
#
# Kept in step with tools/platformio-build.py. Two are load-bearing and are
# explained where they are set: -Wno-comment and the absence of nano.specs.

build.march=rv32imafc_zba_zbb_zbc_zbs_xw
build.mabi=ilp32f

compiler.machine.flags=-march={build.march} -mabi={build.mabi} -msmall-data-limit=8 -msave-restore

# -Wno-comment: the vendor's ch32h417_eth.h has nested "/*" in its register
# banner comments and is included by anything touching the SDK. Left on it
# produces hundreds of lines a build from a header we do not own, which is
# exactly how a real warning gets missed.
compiler.warning_flags=-Wall -Wextra -Wno-comment

compiler.common.flags={compiler.machine.flags} -Os {compiler.warning_flags} -fmessage-length=0 -fsigned-char -ffunction-sections -fdata-sections -fno-common

compiler.c.flags={compiler.common.flags} -std=gnu17 -MMD -c
compiler.cpp.flags={compiler.common.flags} -std=gnu++17 {build.flags.exceptions} -fno-threadsafe-statics -fno-rtti -fno-use-cxa-atexit -MMD -c
compiler.S.flags={compiler.machine.flags} -x assembler-with-cpp -MMD -c
compiler.ar.flags=rcs
compiler.objcopy.eep.flags=-O ihex -j .eeprom --set-section-flags=.eeprom=alloc,load --no-change-warnings --change-section-lma .eeprom=0
compiler.elf2hex.flags=-O binary
compiler.size.flags=-A

# NOTE: --specs=nano.specs must NEVER be added here. It rewrites -lstdc++ to
# -lstdc++_nano, which is built without unwind tables, so every throw reaches
# std::terminate -- after a perfectly clean link, with the failure visible only
# at run time. It is an attractive flag for the smaller printf it buys; that is
# what it costs.
#
# NOT -Wl,--no-relax-gp either: this is GNU ld 2.38, which does not have the
# option, and passing it is a hard link error. If "relocation truncated to fit:
# R_RISCV_GPREL_I" ever appears, the fix is -msmall-data-limit=0.
compiler.c.elf.flags={compiler.machine.flags} -Os -nostartfiles -Wl,--gc-sections -Wl,--print-memory-usage --specs=nosys.specs

compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
compiler.c.elf.extra_flags=
compiler.ar.extra_flags=
compiler.objcopy.eep.extra_flags=
compiler.elf2hex.extra_flags=

# --------------------------------------------------------------------------
# Defines and includes
# --------------------------------------------------------------------------

# Defaults for everything the menus in boards.txt set. A property that only
# ever came from a menu would make this file depend on a particular board
# definition, and a board without that menu would build with the placeholder
# text in its command line rather than fail.

# C++ exceptions, off by default as in arduino-pico and the ESP cores.
# Enabling costs roughly 30 KB of unwind tables, which stay in flash and are
# read only when something throws. One libstdc++.a serves both settings --
# -fno-exceptions only changes codegen for our own sources -- so unlike
# arduino-pico there is no second prebuilt library.
build.flags.exceptions=-fno-exceptions
build.flags.exceptions.define=

# What `Serial` is. Empty means USART1, which is the conservative fallback for
# a board definition with no Serial menu; the boards that have one default to
# USB CDC, which sets -DCH32H4_USB -DUSE_TINYUSB -DCH32H4_SERIAL_IS_USB.
build.flags.serial=

# The filesystem, in bytes, carved out of the flash tail immediately below the
# 16 KB EEPROM. Substituted into the linker script by prebuild hook 1, which
# ASSERTs that the sketch region, the filesystem and the EEPROM tile the 960 KB
# user area exactly. Must be a multiple of the 8 KB erase page; 0 disables it.
build.fs_size=131072

# The clock tree, as one of the SDK's SYSCLK_* macros. There is no sane default
# for this and it is deliberately EMPTY here: cores/ch32h4/ch32h4_clock.c
# refuses to compile without one, because the alternative to a build error is a
# board that silently runs at 25 MHz off the internal RC. Every board sets it.
build.flags.clock=

build.extra_flags=
build.usb_vid=0x1209
build.usb_pid=0x0001
build.usb_manufacturer="WCH"
build.usb_product="CH32H417"

# 0x00008000. The V5F's entry, which main_v3f.c asserts against the linker.
build.v5f_start=0x00008000

build.defines=-DARDUINO=10808 -DARDUINO_ARCH_CH32H4 -DF_CPU={build.f_cpu} -DCH32_V5F_START_ADDR={build.v5f_start} -DARDUINO_{build.board} -DUSB_VID={build.usb_vid} -DUSB_PID={build.usb_pid} '-DUSB_MANUFACTURER={build.usb_manufacturer}' '-DUSB_PRODUCT={build.usb_product}' {build.flags.serial} {build.flags.exceptions.define} {build.flags.clock}

build.sdk.path={runtime.platform.path}/system/ch32h417lib

build.tinyusb.path={runtime.platform.path}/libraries/Adafruit_TinyUSB_Arduino/src

# TWO include sets, and the difference between them is load-bearing.
#
# build.includes.detect is what the core, the variant and the SDK need. It is
# also what arduino-cli uses to work out which libraries a sketch depends on:
# that phase compiles the sketch, reads the "fatal error: X.h: No such file"
# it gets back, finds the library providing X.h, adds it, and repeats.
#
# build.includes adds the TinyUSB headers on top, and is what actually
# compiles. The core needs them -- Arduino.h declares `Serial` as an
# Adafruit_USBD_CDC when USB CDC is selected -- and arduino-cli compiles
# cores/ with the platform's include paths only, never a library's.
#
# Putting them in BOTH sets does not work, and the failure is worth stating
# because it looks like nothing to do with include paths. With the TinyUSB
# headers visible during detection, Arduino.h's #include resolves, the
# detection phase sees no missing header, no library is added, and the link
# fails on `Serial`, dcd_int_handler and tud_task_ext -- symbols from a
# library the build decided it did not need. Held back from detection, the
# same #include fails there, TinyUSB is pulled in, and a sketch gets USB CDC
# without having to include anything itself.
build.includes.detect="-I{build.core.path}" "-I{build.variant.path}" "-I{build.sdk.path}/Core" "-I{build.sdk.path}/Peripheral/inc"
build.includes={build.includes.detect} "-I{build.tinyusb.path}" "-I{build.tinyusb.path}/arduino"

# cores/ch32h4/api is deliberately NOT on the include path. On a
# case-insensitive filesystem it makes <string.h> resolve to ArduinoCore-API's
# own String.h, and every use of strlen, memcpy and memset inside the API then
# fails to compile. Code reaches the API through "api/ArduinoAPI.h", which is
# what arduino-pico does.
build.extra_includes=

compiler.cpreprocessor.flags={build.defines} {build.includes} {build.extra_includes}

# --------------------------------------------------------------------------
# Hooks
# --------------------------------------------------------------------------

# 1. The linker script. One token, substituted from the Filesystem menu.
#    tools/simplesub.py refuses to write a file in which the token was not
#    found, so a renamed placeholder fails here rather than in the linker.
recipe.hooks.prebuild.1.pattern="{ch32h4.python}" "{runtime.platform.path}/tools/simplesub.py" --input "{build.variant.path}/ch32h417.ld" --out "{build.path}/ch32h417.ld" --sub __CH32H4_FS_SIZE__ {build.fs_size}

# 2. The vendor SDK, compiled into an archive. Incremental: the script skips a
#    source whose object is newer, so only the first build pays for it.
#    The flags come last, after a bare "--", and are deliberately NOT
#    wrapped in quotes: {build.defines} contains -DUSB_MANUFACTURER="WCH",
#    whose inner quotes would close an outer pair and hand the script a
#    mangled command line.
recipe.hooks.prebuild.2.pattern="{ch32h4.python}" "{runtime.platform.path}/tools/build_sdk.py" --sdk "{build.sdk.path}" --out "{build.path}/libch32h4sdk.a" --objdir "{build.path}/sdk" --cc "{compiler.path}{compiler.c.cmd}" --ar "{compiler.path}{compiler.ar.cmd}" -- {compiler.c.flags} {compiler.cpreprocessor.flags}

# --------------------------------------------------------------------------
# Recipes
# --------------------------------------------------------------------------

recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.cpreprocessor.flags} {compiler.c.extra_flags} {build.extra_flags} {includes} "{source_file}" -o "{object_file}"

# The library-detection pass. Defined explicitly so it uses
# build.includes.detect; left undefined, arduino-cli derives it from
# recipe.cpp.o.pattern and gets build.includes, which is the case described
# above. -w because this pass exists to produce ONE kind of error and every
# other diagnostic it prints is printed again by the real compile.
recipe.preproc.macros="{compiler.path}{compiler.cpp.cmd}" {compiler.machine.flags} -std=gnu++17 {build.flags.exceptions} -fno-rtti -w -x c++ -E -CC {build.defines} {build.includes.detect} {includes} "{source_file}" -o "{preprocessed_file_path}"

recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.cpreprocessor.flags} {compiler.cpp.extra_flags} {build.extra_flags} {includes} "{source_file}" -o "{object_file}"

recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {compiler.cpreprocessor.flags} {compiler.S.extra_flags} {build.extra_flags} {includes} "{source_file}" -o "{object_file}"

recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} {compiler.ar.extra_flags} "{archive_file_path}" "{object_file}"

# --whole-archive around the core archive: the variant's pin table must win
# over the core's weak defaults, and a plain archive member is only pulled in
# if something already references it.
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} {compiler.c.elf.extra_flags} "-T{build.path}/ch32h417.ld" "-Wl,-Map,{build.path}/{build.project_name}.map" -o "{build.path}/{build.project_name}.elf" "-L{build.path}" -Wl,--start-group {object_files} -Wl,--whole-archive "{archive_file_path}" -Wl,--no-whole-archive "{build.path}/libch32h4sdk.a" -lm -lstdc++ -lc -Wl,--end-group

recipe.objcopy.bin.pattern="{compiler.path}{compiler.elf2hex.cmd}" {compiler.elf2hex.flags} {compiler.elf2hex.extra_flags} "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"

recipe.output.tmp_file={build.project_name}.bin
recipe.output.save_file={build.project_name}.{build.variant}.bin

recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.text|\.rodata|\.itcm_text|\.data)\s+([0-9]+).*
recipe.size.regex.data=^(?:\.data|\.bss)\s+([0-9]+).*

# --------------------------------------------------------------------------
# Upload
# --------------------------------------------------------------------------
#
# wlink, the same tool the PlatformIO platform and the hardware tests use.
# Pinned expectations: 0.1.1 reports "Probe is not attached to an MCU" on this
# part, which is a version problem wearing the costume of a wiring problem.

tools.wlink.cmd=wlink
tools.wlink.path={runtime.tools.wlink.path}

tools.wlink.upload.params.verbose=-v
tools.wlink.upload.params.quiet=-q
tools.wlink.upload.pattern="{path}/{cmd}" flash "{build.path}/{build.project_name}.bin"

tools.wlink.erase.params.verbose=-v
tools.wlink.erase.params.quiet=-q
tools.wlink.erase.pattern="{path}/{cmd}" erase
